% Title-page background: the artwork is A4-landscape (~1.41:1), not 16:9.
% Scale it to the frame height and pin it to the bottom-right corner so the
% swoosh keeps its proportions; the remaining strip on the left stays white.
\setbeamertemplate{background canvas}{
  \ifnum\thepage=1\relax
    \begin{tikzpicture}[remember picture,overlay]
      \node[anchor=south east,inner sep=0pt] at (current page.south east) {
        \includegraphics[height=\paperheight]{background.png}
      };
    \end{tikzpicture}
  \else
    % Bleed: repeat the bar colours slightly past the page edges, under the
    % headline and footline, so no white hairline shows at the edges.
    \begin{tikzpicture}[remember picture,overlay]
      \fill[bluep] ([xshift=-2pt,yshift=2pt]current page.north west)
        rectangle ([yshift=-3pt]current page.north);
      \fill[tealp] ([yshift=-3pt]current page.north)
        rectangle ([xshift=2pt,yshift=2pt]current page.north east);
      \fill[bluep] ([xshift=-2pt,yshift=-2pt]current page.south west)
        rectangle ([xshift=.666667\paperwidth,yshift=3pt]current page.south west);
      \fill[tealp] ([xshift=.666667\paperwidth,yshift=-2pt]current page.south west)
        rectangle ([xshift=2pt,yshift=3pt]current page.south east);
    \end{tikzpicture}
  \fi
}
